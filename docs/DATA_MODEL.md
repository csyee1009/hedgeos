# HedgeOS Data Model & Domain Entities

---

## 1. Domain Model Entities

### `FieldProvenance`
Tracks the exact source and confidence of parsed fields to eliminate implicit assumptions.
```typescript
export type ProvenanceSource = "USER_EXPLICIT" | "AI_INFERRED" | "SYSTEM_DEFAULT";

export interface FieldProvenance<T> {
  value: T;
  source: ProvenanceSource;
  confidence: number; // 0.0 - 1.0
  requiresConfirmation: boolean;
  originalPhrase?: string;
}
```

### `TypedRiskIntent`
The structured financial intent generated by Layer 1 and locked by user confirmation.
```typescript
export interface TypedRiskIntent {
  intentId: string;
  timestamp: number;
  confirmedByUser: boolean;
  asset: FieldProvenance<string>;
  exposureAmount: FieldProvenance<number>;
  targetMaxLossPercent: FieldProvenance<number>;
  maxPremiumUSDC: FieldProvenance<number>;
  horizonTimestamp: FieldProvenance<number>;
  allowedProtocols: FieldProvenance<string[]>;
  allowMultiLeg: FieldProvenance<boolean>;
}
```

### `MarketQuote`
Unified quote output from market sources.
```typescript
export type QuoteSourceType = "OPTION_BOOK" | "RFQ_OPTION_FACTORY";

export interface MarketQuote {
  quoteId: string;
  sourceType: QuoteSourceType;
  protocol: "THETANUTS";
  asset: string;
  optionType: "PUT" | "CALL" | "PUT_SPREAD";
  strikePrice: number;
  expiryTimestamp: number;
  premiumUSDC: number;
  liquidityAvailable: number;
  executableNow: boolean;
  calldata?: string;
}
```

### `CandidateStrategy`
Evaluation outcome of a strategy candidate.
```typescript
export type CandidateStatus = "PASS" | "REJECTED";

export interface CandidateStrategy {
  strategyId: string;
  name: string;
  optionType: string;
  quotes: MarketQuote[];
  status: CandidateStatus;
  rejectionReasons: string[];
  rank?: number;
  scores?: {
    coverageScore: number;
    efficiencyScore: number;
    executabilityScore: number;
    intentFitScore: number;
    compositeScore: number;
  };
  payoffSummary: {
    maxLossPercent: number;
    protectionDurationHours: number;
    totalCostUSDC: number;
    protectedFloorPrice: number;
  };
}
```

### `PolicyDecisionRecord`
Digest-bound verification record from the Financial Constitution.
```typescript
export interface PolicyCheckItem {
  ruleId: string;
  description: string;
  passed: boolean;
  details: string;
}

export interface PolicyDecisionRecord {
  decisionId: string;
  intentId: string;
  strategyId: string;
  passedAllInvariants: boolean;
  checks: PolicyCheckItem[];
  timestamp: number;
}
```

### `SimulationResult`
Simulation status and transaction preview payload.
```typescript
export type SimulationStatus = "NOT_AVAILABLE" | "PREVIEW_ONLY" | "PROVIDER_SIMULATED" | "CHAIN_SIMULATED";

export interface SimulationResult {
  simulationId: string;
  status: SimulationStatus;
  intentId: string;
  expectedAsset: string;
  expectedPremiumUSDC: number;
  expectedExpiry: number;
  targetContract: string;
  bindingHash: string;
  authorizedByHuman: boolean;
}
```
